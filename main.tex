% MODE SWITCH
%
%   \CompendiumMode = 1   full compendium: book class, title page,
%                         table of contents, numbered chapters,
%                         every \include'd chapter
%
%   \CompendiumMode = 0   one chapter on its own, as a lecture note:
%                         article class, plain title + date, no TOC,
%                         sections numbered 1, 2, 3 instead of 3.1, 3.2
%
% In note mode \NoteChapter names the single chapter file to typeset,
% without the .tex extension.
%
% Build:
%   compendium     latexmk -pdf -outdir=build main.tex
%   lecture note   set the two switches below and build as usual, or
%                  leave them alone and override on the command line:
%     latexmk -pdf -outdir=build -jobname=note_lp \
%       -usepretex='\def\CompendiumMode{0}\def\NoteChapter{ch_lp_types}' main.tex
% =====================================================================
